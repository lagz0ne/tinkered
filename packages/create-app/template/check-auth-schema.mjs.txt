import { execFileSync } from "node:child_process";
import { mkdtemp, readFile, rm } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join } from "node:path";

if (import.meta.main) {
  const folder = await mkdtemp(join(tmpdir(), "app-auth-schema-"));
  try {
    const generated = join(folder, "auth-schema.ts");
    execFileSync(
      process.execPath,
      [
        "node_modules/auth/dist/index.mjs",
        "generate",
        "--config",
        "auth.config.ts",
        "--output",
        generated,
        "--yes",
      ],
      { stdio: "pipe" },
    );
    const source = (await readFile(generated, "utf8")).replaceAll("/* @__PURE__ */ ", "");
    const formatted = execFileSync("vp", ["fmt", "--stdin-filepath", "auth-schema.ts"], {
      input: source,
      encoding: "utf8",
    });
    if (formatted !== (await readFile("src/server/auth-schema.ts", "utf8"))) {
      process.stderr.write("Auth schema differs: src/server/auth-schema.ts\n");
      process.exitCode = 1;
    }
  } finally {
    await rm(folder, { recursive: true, force: true });
  }
}
