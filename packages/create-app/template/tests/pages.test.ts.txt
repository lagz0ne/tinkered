import type { PGlite } from "@electric-sql/pglite";
import { createScope, type Scope } from "@tinker/core";
import { createTestDatabase, publishAfterCommit, type TestDatabase } from "@tinker/stack";
import { beforeAll, afterAll, beforeEach, afterEach, expect, test } from "vite-plus/test";
import {
  createPages,
  createWeb,
  databaseConfig,
  migrations,
  published,
  publishNotes,
} from "../src/index.ts";

let template: TestDatabase.Handle;
let client: PGlite;
let scope: Scope.Handle;
let web: ReturnType<typeof createWeb>;
beforeAll(async () => {
  template = await createTestDatabase(migrations);
});
afterAll(async () => {
  await template.close();
});
beforeEach(async () => {
  client = await template.clone();
  const page = createPages({ script: "/assets/client.js", styles: [], dev: false });
  web = createWeb(page.mount);
  scope = createScope({
    tags: [databaseConfig({ client })],
    extensions: [page.extension, web, published, publishAfterCommit(publishNotes)],
  });
  await scope.ready;
});
afterEach(async () => {
  await scope.close();
  await client.close();
});

test("the list page includes saved notes in its first HTML", async () => {
  await scope.resolve(web).request("/api/notes", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ title: "First HTML" }),
  });
  const response = await scope.resolve(web).request("/");
  expect(await response.text()).toContain("<li>First HTML</li>");
});

test.each([
  { path: "/sign-up", title: "Sign up" },
  { path: "/sign-in", title: "Sign in" },
  { path: "/verify", title: "Verify email" },
  { path: "/reset", title: "Reset password" },
  { path: "/new-password", title: "Set new password" },
])("the $path auth page renders on the server", async ({ path, title }) => {
  const response = await scope.resolve(web).request(path);
  expect(await response.text()).toContain(`<h1>${title}</h1>`);
});
