import { createScope } from "@tinker/core";
import { startNatsServer } from "@tinker/nats/testing";
import { createTestDatabase, liveUpdates } from "@tinker/stack";
import { expect, test } from "vite-plus/test";
import { createWeb, databaseConfig, migrations, published, publishNotes } from "../src/index.ts";

test("a saved note reaches another server over NATS and sync", async () => {
  const template = await createTestDatabase(migrations);
  const client = await template.clone();
  const nats = await startNatsServer();
  const first = createWeb();
  const second = createWeb();
  const live = liveUpdates(publishNotes, {
    subject: "__APP_NAME__.changed",
    env: { NATS_URL: nats.url },
  });
  const writer = createScope({
    tags: [databaseConfig({ client })],
    extensions: [first, published, live],
  });
  const viewer = createScope({
    tags: [databaseConfig({ client })],
    extensions: [second, published, live],
  });
  let reader: ReadableStreamDefaultReader<Uint8Array> | undefined;
  try {
    await Promise.all([writer.ready, viewer.ready]);
    const reply = await viewer.resolve(second).request("/sync");
    reader = reply.body?.getReader();
    if (!reader) expect.unreachable("sync needs a response body");
    await reader.read();
    await writer.resolve(first).request("/api/notes", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ title: "Live note" }),
    });
    const next = await reader.read();
    expect(new TextDecoder().decode(next.value)).toContain('"value":[{"title":"Live note"}]');
  } finally {
    await reader?.cancel();
    await Promise.all([writer.close(), viewer.close()]);
    await nats.close();
    await client.close();
    await template.close();
  }
});
