import { spawn } from "node:child_process";
import { mkdtemp, rm } from "node:fs/promises";
import { createServer } from "node:http";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { expect, test } from "vite-plus/test";

test("vp run dev binds local defaults and serves its page", async () => {
  const folder = await mkdtemp(join(tmpdir(), "__APP_NAME__-dev-"));
  const listener = createServer();
  await new Promise<void>((done, fail) => {
    listener.once("error", fail);
    listener.listen(0, "127.0.0.1", done);
  });
  const address = listener.address();
  if (address === null || typeof address === "string") expect.unreachable("a TCP port is needed");
  const port = address.port;
  await new Promise<void>((done, fail) =>
    listener.close((error) => (error ? fail(error) : done())),
  );
  const child = spawn("vp", ["run", "dev"], {
    env: { PATH: process.env.PATH, PORT: String(port), DATA_PATH: join(folder, "db") },
    stdio: ["ignore", "pipe", "pipe"],
    detached: true,
  });
  const lines: string[] = [];
  child.stdout.on("data", (chunk: Buffer) => lines.push(chunk.toString()));
  child.stderr.on("data", (chunk: Buffer) => lines.push(chunk.toString()));
  const closed = new Promise<number | null>((done, fail) => {
    child.once("error", fail);
    child.once("close", done);
  });
  try {
    await expect.poll(() => lines.join(""), { timeout: 20000 }).toContain('"kind":"ready"');
    await fetch(`http://127.0.0.1:${port}/api/notes`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ title: "Dev note" }),
    });
    const reply = await fetch(`http://127.0.0.1:${port}/`);
    expect(await reply.text()).toContain("<li>Dev note</li>");
  } finally {
    if (child.pid === undefined) expect.unreachable("the started child needs a pid");
    process.kill(-child.pid, "SIGTERM");
    await closed;
    await rm(folder, { recursive: true, force: true, maxRetries: 5 });
  }
});
