import { createScope } from "@tinker/core";
import { createJobsClock } from "@tinker/jobs/testing";
import { createTestDatabase } from "@tinker/stack";
import { expect, test } from "vite-plus/test";
import { createJobs, databaseConfig, migrations } from "../src/index.ts";

test("a queued job saves its receipt", async () => {
  const template = await createTestDatabase(migrations);
  const client = await template.clone();
  const clock = await createJobsClock("2030-01-01T08:59:00Z");
  const work = createJobs({});
  const scope = createScope({
    tags: [databaseConfig({ client }), clock.binding],
    extensions: [work.extension],
  });
  try {
    await scope.ready;
    await scope.session((session) =>
      session.run(work.send, { input: { queue: "receipt", data: { title: "Paid" } } }),
    );
    await clock.advance(1000);
    await expect
      .poll(async () => (await client.query("select title from receipts")).rows)
      .toEqual([{ title: "Paid" }]);
  } finally {
    await scope.close({ graceful: true });
    await client.close();
    await template.close();
  }
});
