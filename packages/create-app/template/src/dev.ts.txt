import { fileURLToPath } from "node:url";
import { runDev } from "@tinker/stack/dev";
import { describeError } from "@tinker/stack";

/** Only this dev entry binds local settings. Prod still requires every key. */
export function runDevelopment(env: NodeJS.ProcessEnv, stop: AbortSignal): Promise<number> {
  const host = env.HOST ?? "127.0.0.1";
  const port = env.PORT ?? "4311";
  return runDev(
    {
      root: fileURLToPath(new URL("../", import.meta.url)),
      entry: "src/server/main.ts",
      nats: true,
      env: {
        ...env,
        HOST: host,
        PORT: port,
        DATA_PATH: env.DATA_PATH ?? "./data/__APP_NAME__",
        BETTER_AUTH_SECRET: env.BETTER_AUTH_SECRET ?? "dev-only-secret-with-at-least-32-characters",
        BETTER_AUTH_URL: env.BETTER_AUTH_URL ?? `http://${host}:${port}`,
        OTEL_EXPORTER_OTLP_ENDPOINT: env.OTEL_EXPORTER_OTLP_ENDPOINT ?? "http://127.0.0.1:9",
        OTEL_SERVICE_NAME: env.OTEL_SERVICE_NAME ?? "__APP_NAME__",
      },
      report(event) {
        const line =
          event.kind === "error" ? { kind: "error", ...describeError(event.error) } : event;
        process.stdout.write(`${JSON.stringify(line)}\n`);
      },
    },
    stop,
  );
}

if (import.meta.main) {
  const stop = new AbortController();
  const abort = () => stop.abort();
  process.on("SIGINT", abort);
  process.on("SIGTERM", abort);
  try {
    process.exitCode = await runDevelopment(process.env, stop.signal);
  } finally {
    process.removeListener("SIGINT", abort);
    process.removeListener("SIGTERM", abort);
  }
}
