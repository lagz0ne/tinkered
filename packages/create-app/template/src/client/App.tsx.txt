import { useData, useRun } from "@tinker/react";
import { noteList, pagePath, readNote } from "../shared/notes.ts";
import { AuthForm } from "./AuthForm.tsx";
import { saveNote } from "./actions.ts";

export function App() {
  const notes = useData(noteList);
  const save = useRun(saveNote);
  const path = useData(pagePath);
  if (path === "/sign-up") return <AuthForm mode="sign-up" />;
  if (path === "/sign-in") return <AuthForm mode="sign-in" />;
  if (path === "/reset") return <AuthForm mode="reset" />;
  if (path === "/new-password") return <AuthForm mode="new-password" />;
  if (path === "/verify")
    return (
      <main>
        <h1>Verify email</h1>
        <p>Open the verify link in your mail. Then sign in.</p>
        <a href="/sign-in">Sign in</a>
      </main>
    );
  return (
    <main>
      <h1>Notes</h1>
      <nav>
        <a href="/sign-in">Sign in</a> · <a href="/sign-up">Sign up</a>
      </nav>
      <form
        onSubmit={(event) => {
          event.preventDefault();
          save.run({
            input: readNote({ title: new FormData(event.currentTarget).get("title") }),
          });
        }}
      >
        <label>
          Title
          <input name="title" required />
        </label>
        <button disabled={save.isPending}>Save note</button>
        {save.isError ? <p role="alert">The note could not be saved.</p> : null}
      </form>
      <ul>
        {notes.map((note) => (
          <li key={note.title}>{note.title}</li>
        ))}
      </ul>
    </main>
  );
}
