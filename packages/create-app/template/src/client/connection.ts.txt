import { resource, tag } from "@tinker/core";
import { createSseClient, type Sse } from "@tinker/sync/sse";

export const openSource = tag<(url: string) => Sse.Source>({ label: "sync.openSource" });
export const wire = resource({
  label: "sync.wire",
  depends: { open: openSource },
  factory({ open }, ctx) {
    const transport = createSseClient({ open: () => open("/sync") });
    ctx.defer(() => transport.close());
    return transport;
  },
});
