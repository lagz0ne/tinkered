import { operation } from "@tinker/core";
import { asc } from "drizzle-orm";
import { readNote, noteList } from "../shared/notes.ts";
import { notes } from "./schema.ts";
import { database, transaction } from "./store.ts";

export const addNote = operation({
  label: "notes.add",
  input: readNote,
  depends: { tx: transaction },
  async run({ tx }, { input }) {
    await tx.insert(notes).values(input).onConflictDoNothing();
    return input;
  },
});

export const listNotes = operation({
  label: "notes.list",
  depends: { tx: transaction },
  run: ({ tx }) => tx.select().from(notes).orderBy(asc(notes.title)),
});

export const publishNotes = operation({
  label: "notes.publish",
  depends: { db: database, notes: noteList.controller },
  async run({ db, notes: cell }) {
    cell.set(await db.select().from(notes).orderBy(asc(notes.title)));
  },
});
