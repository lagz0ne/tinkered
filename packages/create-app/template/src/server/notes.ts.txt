import { operation } from "@tinker/core";
import { asc } from "drizzle-orm";
import { raise } from "../errors.ts";
import { notes } from "./schema.ts";
import { transaction } from "./store.ts";

export declare namespace Notes {
  type Note = { title: string };
}

function readNote(input: unknown): Notes.Note {
  if (
    typeof input !== "object" ||
    input === null ||
    !("title" in input) ||
    typeof input.title !== "string" ||
    input.title.trim() === ""
  ) {
    raise("BadNote", { input });
  }
  return { title: input.title.trim() };
}

export const addNote = operation({
  label: "notes.add",
  input: readNote,
  depends: { tx: transaction },
  async run({ tx }, { input }) {
    await tx.insert(notes).values(input).onConflictDoNothing();
    return input;
  },
});

export const listNotes = operation({
  label: "notes.list",
  depends: { tx: transaction },
  run: ({ tx }) => tx.select().from(notes).orderBy(asc(notes.title)),
});
