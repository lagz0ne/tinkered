import { operation, type Operation } from "@tinker/core";
import { emit, route, stream } from "@tinker/hono";
import { source, type Sync } from "@tinker/sync";
import { createSseServer } from "@tinker/sync/sse";
import { describeError } from "@tinker/stack";
import { noteList } from "../shared/notes.ts";

export const published = source({ cells: [[noteList, "notes"]] });

const openWire = operation({
  label: "sync.open",
  depends: { published },
  run: () => ({ type: "register" as const, keys: ["notes"] }),
});
const syncBody = operation({
  label: "sync.body",
  depends: { emit: emit.required, published },
  async run({ emit, published }, ctx: Operation.Ctx<Sync.Message>) {
    const wire = createSseServer(emit, ctx.signal);
    const connected = published.connect(wire);
    wire.deliver(ctx.input);
    const result = await connected;
    if (result.status === "failed") ctx.log.error("sync failed", describeError(result.error));
  },
});

export const syncRoute = route.get("/sync", openWire, {
  respond(register, c) {
    c.header("Content-Type", "text/event-stream");
    c.header("Cache-Control", "no-cache");
    return stream(c, syncBody, { input: register });
  },
});
