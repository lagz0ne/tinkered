import type { Dev } from "@tinker/stack/dev";
import { resolve } from "node:path";
import { createScope, extension, type Observe } from "@tinker/core";
import { liveUpdates, migrate, readExitCode, server, traceSink } from "@tinker/stack";
import { raise } from "../errors.ts";
import { database, databaseConfig } from "./store.ts";
import { migrations } from "./migrations.ts";
import { publishNotes } from "./notes.ts";
import { published } from "./sync.ts";
import { createIdentity } from "./auth.ts";
import { createMail } from "./mail.ts";
import { createJobs } from "./jobs.ts";
import { createWeb } from "./routes.ts";

export async function runServer(
  env: NodeJS.ProcessEnv,
  stop: AbortSignal,
  host?: Dev.Wiring,
): Promise<number> {
  let log: Observe.Logger | undefined;
  const traces = traceSink({
    env: {
      OTEL_EXPORTER_OTLP_ENDPOINT: env.OTEL_EXPORTER_OTLP_ENDPOINT,
      OTEL_SERVICE_NAME: env.OTEL_SERVICE_NAME,
    },
    write: (line) => process.stdout.write(`${line}\n`),
  });
  const post = createMail(env);
  const work = createJobs(env, [post.job]);
  const identity = createIdentity(work, post);
  const entry = host
    ? await host.load<typeof import("./pages.tsx")>("src/server/pages.tsx")
    : await import("./pages.tsx");
  const page = entry.createPages(
    host
      ? { script: "/src/client/boot.tsx", styles: ["/src/client/style.css?direct"], dev: true }
      : { script: "/assets/client.js", styles: ["/assets/client.css"], dev: false },
  );
  const web = createWeb(page.mount, identity.wiring);
  const scope = createScope({
    signal: stop,
    tags: [
      databaseConfig(host ? { client: host.client } : env.DATA_PATH),
      host && post.backend("log"),
      identity.config({
        BETTER_AUTH_SECRET: env.BETTER_AUTH_SECRET,
        BETTER_AUTH_URL: env.BETTER_AUTH_URL,
      }),
    ],
    observe: traces.observe,
    extensions: [
      extension({
        label: "__APP_NAME__.root",
        hooks: {
          start(event) {
            log = event.log;
            return event.next();
          },
        },
      }),
      host ? [] : server(web, { env, clientDir: resolve("dist/client") }),
      traces.extension,
      extension({
        label: "__APP_NAME__.settings",
        hooks: {
          start(event) {
            if (!env.DATA_PATH) raise("BadDataSettings", { keys: ["DATA_PATH"] });
            return event.next();
          },
        },
      }),
      migrate(database, migrations),
      post.extension,
      work.extension,
      identity.extension,
      page.extension,
      web,
      published,
      liveUpdates(publishNotes, { subject: "__APP_NAME__.changed", env, connection: host?.connection }),
    ],
  });
  host?.ready(scope.ready.then(() => scope.resolve(web)));
  const phase = await scope.ready.then(
    () => "shutdown" as const,
    () => "boot" as const,
  );
  return readExitCode(await scope.closed, log, phase);
}

if (import.meta.main) {
  const stop = new AbortController();
  process.once("SIGINT", () => stop.abort());
  process.once("SIGTERM", () => stop.abort());
  process.exitCode = await runServer(process.env, stop.signal);
}
