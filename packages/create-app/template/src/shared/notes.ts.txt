import { data } from "@tinker/core";
import { raise } from "../errors.ts";

export declare namespace Notes {
  type Note = { title: string };
}

/** Both HTTP input and sync snapshots cross a process boundary. */
export function readNote(input: unknown): Notes.Note {
  if (
    typeof input !== "object" ||
    input === null ||
    !("title" in input) ||
    typeof input.title !== "string" ||
    input.title.trim() === ""
  ) {
    raise("BadNote", { input });
  }
  return { title: input.title.trim() };
}

export const noteList = data<Notes.Note[]>({
  label: "notes",
  initial: [],
  parse: readNotes,
  eq: (a, b) => JSON.stringify(a) === JSON.stringify(b),
});

export function readNotes(input: unknown): Notes.Note[] {
  if (!Array.isArray(input)) raise("BadNote", { input });
  return input.map(readNote);
}
