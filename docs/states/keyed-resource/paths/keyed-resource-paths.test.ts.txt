/**
 * Replays of the bug paths that the TLA+ model of ADR 0157 found
 * (/home/paseo/statechart/keyed-resource/report.md). One step per action.
 * A reader is what React's useData does:
 *   render    = scope.resolve(read), then reader.get()
 *   subscribe = reader.watch(fn)
 *   leave     = the stop that watch returned
 * Each `expect` is the ADR rule. A failing test means the bug is real.
 */
import { expect, test } from "vite-plus/test";

import { createScope, data, resource, tag } from "@/main";

const key = tag<string>({ label: "sc.key" });

/** One keyed resource on one source. Build n defers "end n"; it throws when `fail` is set. */
function setup(label: string) {
  const src = data({ label: `${label}.src`, initial: 0 });
  const log: string[] = [];
  const control = { fail: false, builds: 0 };
  const unit = resource({
    label,
    config: { k: key.required },
    depends: { src },
    factory: ({ src }, { defer }) => {
      const n = ++control.builds;
      log.push(`build ${n}`);
      if (control.fail) throw new Error(`build ${n} throws`);
      defer(() => void log.push(`end ${n}`));
      return src;
    },
  });
  return { src, log, control, unit };
}

/** A render: resolve, then get. A fault is the reader's to show; the test only steps. */
function renders(scope: ReturnType<typeof createScope>, read: unknown) {
  const reader = scope.resolve(read as never) as { get(): unknown; watch(fn: () => void): () => void };
  try {
    reader.get();
  } catch {
    /* the entry holds a fault */
  }
  return reader;
}

test("NoChangeNoBuild + OneEntryPerKey (ADR 0157 d4, ADR 0150 d5): a reader that subscribes after the last reader left", () => {
  const { log, control, unit } = setup("sc.one");
  const scope = createScope();

  // 1. reader 1 renders key A [no entry: a new one]; 2. the build throws
  control.fail = true;
  const one = renders(scope, unit({ k: "A" }));
  control.fail = false;
  // 3. reader 2 renders key A [the entry is there]
  const two = renders(scope, unit({ k: "A" }));
  expect(two).toBe(one);
  // 4. reader 2 subscribes
  const stop = two.watch(() => undefined);
  // 5. reader 2 leaves (reader 1 rendered and still holds the entry)
  stop();
  // 6. reader 2 renders key A again
  const again = renders(scope, unit({ k: "A" }));

  // ADR 0150 d5: one entry per key text; every reader of the same key shares it.
  // ADR 0157 d4: a read with no change makes no build.
  expect({ same: again === one, log }).toEqual({ same: true, log: ["build 1"] });
});

test("EndBeforeNextBuild (ADR 0157 d5): a reader that left subscribes again (StrictMode)", () => {
  const { src, log, control, unit } = setup("sc.two");
  const scope = createScope();

  // 1. reader 1 renders key A; 2. the build throws
  control.fail = true;
  const one = renders(scope, unit({ k: "A" }));
  control.fail = false;
  // 3. reader 1 subscribes
  const stop = one.watch(() => undefined);
  // 4. reader 1 leaves
  stop();
  // 5. reader 2 renders key A [no entry: a new one]; 6. the build finishes (defers "end 2")
  const two = renders(scope, unit({ k: "A" }));
  expect(two).not.toBe(one);
  // 7. the source changes
  scope.controller(src).set(1);
  // 8. reader 1 subscribes again to the entry it holds
  one.watch(() => undefined);

  // ADR 0157 d5: an end runs before the next build of the key.
  const build3 = log.indexOf("build 3");
  expect(build3).toBeGreaterThan(-1);
  expect(log.slice(0, build3)).toContain("end 2");
});
