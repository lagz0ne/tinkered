import { createMiddleware } from "@tanstack/react-start";
import { extension } from "@tinker/core";
import type { Scope } from "@tinker/core";
import { requestHeaders } from "../backend/index.ts";
import { raise } from "../errors.ts";
import { holdResponse } from "./body.server.ts";

/** Start merges this registry to type the context supplied by the server entry. */
declare module "@tanstack/react-start" {
  interface Register {
    server: { requestContext: { scope?: Scope.Handle } };
  }
}

/** This root middleware uses only fetch context, before global middleware has run. */
const middleware = createMiddleware().server(async ({ context, request, next }) => {
  const scope = context.scope;
  if (scope === undefined) raise("StartScopeMissing", {});
  const session = scope.createSession({ tags: requestHeaders(new Headers(request.headers)) });
  let closed: Promise<void> | undefined;
  const finish = (graceful: boolean) =>
    (closed ??= session.close({ graceful }).then((end) => {
      if (end.teardownErrors?.length) throw end.teardownErrors.at(0);
    }));
  try {
    const result = await next({ context: { session, signal: request.signal } });
    return { ...result, response: await holdResponse(result.response, finish) };
  } catch (error) {
    await finish(false);
    throw error;
  }
});

/** Startup supplies the native fetch binding; all Start boundaries share this middleware. */
export const startRequests = Object.assign(
  extension({
    label: "start.requests",
    hooks: {
      async start(event) {
        await event.next();
        return { scope: event.scope };
      },
    },
  }),
  { middleware },
);

import { createStartHandler as createStartFetch } from "@tanstack/react-start/server";
import { defaultStreamHandler as renderStartStream } from "@tanstack/react-start/server";
import type { getRouter } from "./router.tsx";
import { holdResponse } from "./transport/body.server.ts";
import { getBackend, closeBackend } from "./transport/entry.server.ts";
const renderRequest = createStartFetch(async (context) => {
  const router = context.router as ReturnType<typeof getRouter>;
  try {
    const output = await renderStartStream(context);
    if (output instanceof Response) return holdResponse(output, () => router.close());
    return { ...output, response: await holdResponse(output.response, () => router.close()) };
  } catch (error) {
    await router.close();
    throw error;
  }
});
export default {
  async fetch(request: Request) {
    const { requestContext } = await getBackend();
    return renderRequest(request, { context: requestContext });
  },
};
export const close = closeBackend;

import { createScope, extension } from "@tinker/core";
import { tanstackStartCookies } from "better-auth/tanstack-start";
import { authSettings, databaseSettings, mailSettings, migrate } from "../backend/index.ts";
import { history, observer } from "../telemetry/index.ts";
import { startRequests } from "./start.ts";
import { readSettings } from "./settings.server.ts";
const setup = extension({
  label: "backend.setup",
  hooks: {
    async start(event) {
      await event.next();
      await event.scope.run(migrate);
    },
  },
});
let owned: ReturnType<typeof start> | undefined;
/** This framework entry retains one process root; importing it starts no clients. */
async function start() {
  const settings = readSettings(process.env);
  const toolStop = new AbortController();
  const tools = createScope({ signal: toolStop.signal });
  await tools.ready;
  const stop = new AbortController();
  const app = createScope({
    signal: stop.signal,
    observe: tools.resolve(observer),
    extensions: [setup, startRequests],
    tags: [
      databaseSettings(settings.database),
      mailSettings(settings.mail),
      authSettings({
        origin: settings.origin,
        secret: settings.secret,
        plugins: [tanstackStartCookies()],
      }),
    ],
  });
  try {
    await app.ready;
  } catch (error) {
    toolStop.abort();
    await tools.closed;
    throw error;
  }
  let closed: Promise<void> | undefined;
  const close = () =>
    (closed ??= (async () => {
      stop.abort();
      const end = await app.closed;
      toolStop.abort();
      await tools.closed;
      if (end.status === "failed") throw end.error;
      if (end.teardownErrors?.length) throw end.teardownErrors.at(0);
    })());
  if (import.meta.hot) import.meta.hot.dispose(close);
  return {
    requestContext: app.resolve(startRequests),
    close,
    readHistory: () => tools.resolve(history),
  };
}
export function getBackend() {
  return (owned ??= start());
}
export async function closeBackend() {
  if (owned) await (await owned).close();
}

import { operation } from "@tinker/core";
import { eq } from "drizzle-orm";
import { principal } from "./auth.ts";
import { database, transaction } from "./database.ts";
import { user } from "./schema.ts";
import { readProfileInput } from "../contracts/profile.ts";
import type { Profile } from "../contracts/profile.ts";
import { raise } from "../errors.ts";
export const authorize = operation({
  label: "authorize",
  depends: { principal },
  run: async ({ principal }) => {
    if (principal === null) raise("SignInRequired", {});
    return principal.id;
  },
});
export const readProfile = operation({
  label: "readProfile",
  depends: { principal, database },
  run: async ({ principal, database }): Promise<Profile.Value | null> => {
    if (principal === null) return null;
    return (
      (
        await database
          .select({
            id: user.id,
            name: user.name,
            email: user.email,
            emailVerified: user.emailVerified,
          })
          .from(user)
          .where(eq(user.id, principal.id))
      ).at(0) ?? null
    );
  },
});
export const writeProfile = operation({
  label: "writeProfile",
  depends: { transaction },
  run: async (
    { transaction },
    ctx: import("@tinker/core").Operation.Ctx<{ id: string; name: string }>,
  ) => {
    const saved = (
      await transaction
        .update(user)
        .set({ name: ctx.input.name, updatedAt: new Date(ctx.clock.currentTimeMillis()) })
        .where(eq(user.id, ctx.input.id))
        .returning({
          id: user.id,
          name: user.name,
          email: user.email,
          emailVerified: user.emailVerified,
        })
    ).at(0);
    if (saved === undefined) raise("SignInRequired", {});
    ctx.log("profile.saved");
    return saved;
  },
});
export const saveProfile = operation({
  label: "saveProfile",
  input: readProfileInput,
  depends: { authorize, write: writeProfile },
  run: async ({ authorize, write }, ctx) => {
    const id = await authorize.run();
    return write.run({ input: { id, name: ctx.input.name } });
  },
});

import { transferResponseBodyOwnership } from "@tanstack/react-start/server";
/** Retains request work until the consumer ends, fails, or cancels the body. */
export async function holdResponse(
  response: Response,
  finish: (graceful: boolean) => Promise<void>,
): Promise<Response> {
  if (response.body === null) {
    await finish(true);
    return response;
  }
  const reader = response.body.getReader();
  let cancelled = false;
  const body = new ReadableStream<Uint8Array>({
    async pull(controller) {
      try {
        const part = await reader.read();
        if (part.done) {
          await finish(!cancelled);
          if (!cancelled) controller.close();
        } else controller.enqueue(part.value);
      } catch (error) {
        await finish(false);
        controller.error(error);
      }
    },
    async cancel(reason) {
      cancelled = true;
      const results = await Promise.allSettled([finish(false), reader.cancel(reason)]);
      for (const result of results) {
        if (result.status === "rejected") throw result.reason;
      }
    },
  });
  return transferResponseBodyOwnership(
    response,
    new Response(body, {
      status: response.status,
      statusText: response.statusText,
      headers: response.headers,
    }),
  );
}
